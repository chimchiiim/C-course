Урок 1

Основы ввода-вывода с printf и scanf_s

Зачем нужны безопасные версии (_s)?

Стандартная функция scanf("%s", buffer) считывает символы до пробела без проверки размера выделенной памяти. Если ввод превысит размер массива, произойдет переполнение буфера (Buffer Overflow) — классическая уязвимость безопасности C/C++.

Функции с суффиксом _s (от англ. safe) требуют указания максимального размера принимаемого буфера для строк и символов.

Основные спецификаторы форматирования

Спецификатор Тип данных Пример printf Пример scanf_s
%d / %i int (целое число) printf("%d", cnt); scanf_s("%d", &cnt);
%lf double (вещественное) printf("%.2lf", mean); scanf_s("%lf", &x);
%f float (одинарная точность) printf("%f", val); scanf_s("%f", &val);
%8.2lf double с шириной и точностью printf("%8.2lf", a[i]); —
%s char[] (строка) printf("%s", str); scanf_s("%s", str, 20);

Урок 2

Подробный разбор C++ кода и файлового ввода-вывода

main.cpp
MSVC C++ Standard 
#include <cstdio>
#include <cmath>
#include <locale.h>

const int max_size = 20;                 // Максимальное количество элементов массива

int main(int argc, char* argv[])
{
   // Входные данные
   double a[max_size];                  // Исходный массив A
   int size;                            // Количество элементов массива
   double x;                            // Заданное число

   // Выходные данные
   bool exists;                         // Признак наличия элемента, меньшего x
   int np;                              // Индекс первого элемента, меньшего x
   double mean;                         // Среднее арифметическое ненулевых элементов
   int cnt;                             // Количество неотрицательных элементов
   int imax;                            // Индекс наибольшего из значений sqrt(e^A) - A^2

   // Промежуточные данные
   double sum;                          // Сумма ненулевых элементов
   int nz;                              // Количество ненулевых элементов
   double f, fmax;                      // Текущее и наибольшее значения функции
   FILE* file;                          // Файловая переменная

   setlocale(LC_ALL, "rus");

   if (argc < 3)                                       // Проверка количества параметров
       printf("Недостаточно параметров!\n");
   else if (fopen_s(&file, argv[1], "r") != 0)         // Проверка открытия входного файла
       printf("Невозможно открыть файл '%s' для чтения\n", argv[1]);
   else
   {
       // Ввод исходных данных
       fscanf_s(file, "%d", &size);
       if (size < 1 || size > max_size)
       {
           fclose(file);
           printf("Ошибка: количество элементов должно быть от 1 до %d\n", max_size);
       }
       else
       {
           for (int i = 0; i < size; i++)
               fscanf_s(file, "%lf", a + i);
           fscanf_s(file, "%lf", &x);
           fclose(file);

           // Решение задачи
           // Поиск первого элемента, меньшего x (досрочный выход)
           for (np = 0; np < size && a[np] >= x; np++)
               ;
           exists = (np < size);

           sum = 0; nz = 0; cnt = 0; mean = 0; imax = 0;
           if (exists)
           {
               // Среднее ненулевых и число неотрицательных
               for (int i = 0; i < size; i++)
               {
                   if (a[i] != 0)
                   {
                       sum += a[i];
                       nz++;
                   }
                   if (a[i] >= 0)
                       cnt++;
               }
               if (nz > 0)
                   mean = sum / nz;
           }
           else
           {
               // Номер наибольшего из значений sqrt(e^A) - A^2
               fmax = sqrt(exp(a[0])) - a[0] * a[0];
               for (int i = 1; i < size; i++)
               {
                   f = sqrt(exp(a[i])) - a[i] * a[i];
                   if (f > fmax)
                   {
                       fmax = f;
                       imax = i;
                   }
               }
           }

           // Вывод результатов
           if (fopen_s(&file, argv[2], "w") != 0)
               printf("Невозможно открыть файл '%s' для записи\n", argv[2]);
           else
           {
               fprintf(file, "Массив A из %d элементов\n", size);
               for (int i = 0; i < size; i++)
                   fprintf(file, "%8.2lf ", a[i]);
               fprintf(file, "\n");
               fprintf(file, "Заданное число x = %.2lf\n", x);
               if (exists)
               {
                   fprintf(file, "В массиве A есть элемент, меньший x, его номер = %d\n", np + 1);
                   if (nz > 0)
                       fprintf(file, "Среднее арифметическое ненулевых элементов = %.2lf\n", mean);
                   else
                       fprintf(file, "Ненулевых элементов нет, среднее не определено\n");
                   fprintf(file, "Количество неотрицательных элементов = %d\n", cnt);
               }
               else
               {
                   fprintf(file, "В массиве A нет элементов, меньших x\n");
                   fprintf(file, "Номер наибольшего из значений sqrt(e^A) - A^2 = %d\n", imax + 1);
               }
               fclose(file);
           }
       }
   }
   return 0;
}

Детальный разбор логики по блокам

Блок 1

Аргументы командной строки

Проверка argc < 3 гарантирует, что программе переданы имена входного и выходного файлов.
argv[0] — путь к exe,
argv[1] — входной файл ("in.txt"),
argv[2] — выходной файл ("out.txt").

Блок 2

Безопасное открытие файлов fopen_s

В отличие от устаревшей fopen, безопасная версия fopen_s(&file, path, mode) принимает указатель на указатель и возвращает код ошибки (0 при успехе).

Блок 3

Чтение fscanf_s и валидация

Код сначала считывает количество элементов size и проверяет его: size > max_size (20). Это исключает выход за границы физического массива double a[20].

Блок 4

Ветвление алгоритма (exists)

Ветка A (exists=true): Находится среднее ненулевых mean = sum/nz и количество неотрицательных cnt.
Ветка B (exists=false): Находится индекс элемента с максимальным результатом формулы f = sqrt(exp(a[i])) - a[i]*a[i].
